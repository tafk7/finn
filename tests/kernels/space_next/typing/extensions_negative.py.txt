from finn.kernels.space._next import (
    Param, ScopeBuilder, Space, ValueKey, ViewKey, Answer, Decided, ValueSemantics,
)

builder = ScopeBuilder(Space)
number = builder.param("number", int)
text = builder.param("text", str)
view = builder.view("complete", number)
builder.bind(number, "wrong")  # E
builder.export(ValueKey("number", int)).value(text)  # E
builder.export(ViewKey("complete", str)).view(view)  # E
builder.export(ValueKey("value", int)).value(view)  # E
builder.export(ViewKey("view", int)).view(number)  # E

def wrong(*, number: int) -> Answer[str]:
    return Decided(str(number))

builder.derived("wrong", wrong, semantics=ValueSemantics.immutable_nominal(int))  # E
