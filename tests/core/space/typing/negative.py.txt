from finn.core.space import UNSUPPLIED, Decision, Param, Space, View, constraint, derived, view, ValueSemantics, QueryResult, Available, configure

class Child(Space):
    width: Param[int] = Param(int)
    label: Param[str] = Param(str, default=UNSUPPLIED)
    lanes = Decision(int, values=(1, 2))

    @view
    def physical(self) -> int:
        return self.width

class Boiler(Space):
    kw: Param[int] = Param(int)

class HeatPump(Space):
    kw: Param[int] = Param(int)
    cop = Decision(int, values=(3, 4))

class Stream(Space):
    spec: Param[int] = Param(int)

class Producer(Space):
    output: Param[Stream] = Param(Stream)

    @derived
    def wrong_spec(self) -> str:
        return self.output.spec  # E

class Parent(Space):
    child = Child(width=4)
    bare = Child()  # not an error: an assignment may supply the width
    bare.width = "four"  # E
    bare.depth = 2  # E
    edge = Stream(spec=1)
    producer = Producer(output=child)  # E
    later = Producer()
    later.output = child  # E
    later.output = edge
    wrong_literal = Child(width="four")  # E
    unknown = Child(width=1, depth=2)  # E
    wrong_reference = Child(width=child.label)  # E
    wrong_guard = Child(width=1, when=child.width)  # E
    heating = Decision[Boiler | HeatPump](values={"boiler": Boiler(kw=1), "heat_pump": HeatPump(kw=2)})
    not_on_every_candidate = Child(width=heating.cop)  # E
    joined = Decision(values={"boiler": Boiler(kw=1), "heat_pump": HeatPump(kw=2)})
    joined_member = Child(width=joined.kw)  # E

    @derived
    def invalid_child_read(self) -> str:
        return self.child.width  # E

    @derived
    def invalid_scalar_read(self) -> str:
        return self.child.lanes  # E

def bad(point: Child, parent: Parent) -> None:
    wrong_value: str = point.width  # E
    wrong_descriptor: int = Child.width  # E
    wrong_configuration: str = configure(Child(width=1))  # E
    point.field(Child.lanes).change("two")  # E
    point.field(Child.width).change(2)  # E
    point.physical(3)  # E
    wrong_assessment: int = point.physical.inspect()  # E
    wrong_output: str = point.physical()  # E
    point.physical.assess()  # E
    point.try_with_choices('not a change')  # E

@constraint  # E
def wrong_constraint() -> str:
    return "yes"

@derived(semantics=ValueSemantics.immutable_nominal(int))  # E
def wrong_answer() -> QueryResult[str]:
    return Available("bad")

bad_guard = Decision(int, values=(1,), when=Child.width)  # E
bad_view = View(Child.width, when=Child.width)  # E

@derived(when=Child.width)  # E
def wrong_derived_guard() -> int:
    return 1
