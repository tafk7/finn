from finn.core.space import (
    Decision, Param, Space, Subspace, ValueKey, ViewKey, constraint, derived, view,
    ValueSemantics, QueryResult, Available, View, SubspaceChoice, refinement,
)

class Child(Space):
    width = Param(int)
    lanes = Decision(int, values=(1, 2))

    @view
    def physical(*, width: int) -> int:
        return width

class Parent(Space):
    child = Subspace(Child, width=4)

def bad(point: Child, parent: Parent) -> None:
    wrong_value: str = point.width  # E
    wrong_descriptor: int = Child.width  # E
    point.field(Child.lanes).change("two")  # E
    refinement.change(point, Child.lanes, "two")  # E
    point.field(Child.width).change(2)  # E
    parent.field(Parent.child.ref(Child.lanes)).change(2)  # E
    point.physical(3)  # E
    wrong_assessment: int = point.physical()  # E
    Parent.child.accepted(Child.width)  # E
    Parent.child.ref(ViewKey("view", int))  # E
    Parent.child.accepted(ValueKey("value", int))  # E
    refinement.commit(point, "not a change")  # E

@constraint  # E
def wrong_constraint() -> str:
    return "yes"

@derived(semantics=ValueSemantics.immutable_nominal(int))  # E
def wrong_answer() -> QueryResult[str]:
    return Available("bad")

bad_guard = Decision(int, values=(1,), when=Child.width)  # E
bad_child = Subspace(Child, width=4, when=Child.width)  # E
bad_choice = SubspaceChoice({"one": Subspace(Child, width=4)}, when=Child.width)  # E
bad_view = View(Child.width, when=Child.width)  # E

@derived(when=Child.width)  # E
def wrong_derived_guard() -> int:
    return 1

@view(when=Child.width)  # E
def wrong_view_guard() -> int:
    return 1

@constraint(when=Child.width)  # E
def wrong_constraint_guard() -> bool:
    return True
