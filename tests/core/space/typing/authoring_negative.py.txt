from dataclasses import dataclass

from finn.core.space import (
    Available,
    Decision,
    Param,
    QueryResult,
    Select,
    Space,
    constraint,
    derived,
    design_space,
    view,
)


class Child(Space):
    width: int = Param()
    label: str | None = Param(default=None)
    lanes: int = Decision(values=(1, 2))

    @view
    def physical(self) -> int:
        return self.width


class Boiler(Space):
    kw: int = Param()


class HeatPump(Space):
    kw: int = Param()
    cop: int = Decision(values=(3, 4))


@dataclass(frozen=True)
class Spec:
    lanes: int


class Producer(Space):
    spec: Spec = Param()

    @derived
    def wrong_spec(self) -> str:
        return self.spec.lanes  # E


class Parent(Space):
    child = Child(width=4)
    bare = Child()  # not an error: an assignment may supply the width
    bare.width = "four"  # E
    bare.depth = 2  # E
    producer = Producer(spec=child)  # E
    wrong_literal = Child(width="four")  # E
    unknown = Child(width=1, depth=2)  # E
    wrong_reference = Child(width=child.label)  # E
    wrong_guard = Child(width=1, when=child.width)  # E
    heating: Boiler | HeatPump = Select(boiler=Boiler(kw=1), heat_pump=HeatPump(kw=2))
    not_on_every_candidate = Child(width=heating.cop)  # E
    joined: Boiler | HeatPump = Select(boiler=Boiler(kw=1), heat_pump=HeatPump(kw=2))
    joined_member = Child(width=joined.kw)  # not an error: the annotation types the choice
    from_view = Child(width=child.physical)  # not an error: a view reference is its value
    view_called = Child(width=child.physical())  # E
    view_wrong_type = Child(label=child.physical)  # E

    @derived
    def invalid_child_read(self) -> str:
        return self.child.width  # E

    @derived
    def invalid_scalar_read(self) -> str:
        return self.child.lanes  # E

    @view
    def own(self) -> int:
        return 1

    own_supplied = Child(width=own)  # E


def bad(point: Child, parent: Parent) -> None:
    wrong_value: str = point.width  # E
    wrong_descriptor: str = Child.width  # E
    wrong_configuration: str = design_space(Child(width=1))  # E
    point.field(Child.lanes).change("two")  # E
    point.field(Child.width).change(2)  # not an error: a Param key types as a Decision's
    point.physical(3)  # E
    point.physical()  # E
    wrong_assessment: int = point.inspect(Child.physical)  # E
    wrong_view_query: QueryResult[str] = point.query(Child.physical)  # E
    wrong_output: str = point.physical  # E
    point.physical.inspect()  # E
    point.view(Child.physical)  # E
    point.inspect(Child.width)  # not an error: a key typed as its value (refused at runtime)
    point.try_with_choices("not a change")  # E
    point.present(3)  # not an error: a value input types as its value (refused at runtime)
    wrong_query: QueryResult[int] = parent.producer.query(Producer.spec)  # E


@constraint  # E
def wrong_constraint() -> str:
    return "yes"


@derived
def wrong_answer() -> QueryResult[str]:
    return Available("bad")


wrong_answer_type: int = wrong_answer  # E

bad_guard: int = Decision(values=(1,), when=Child.width)  # E


class Holder(Space):
    part = Child(width=1)


class Kernel(Space):
    spec: Spec = Param()
    spec_typo = Child(width=spec.lanez)  # E
    spec_wrong = Child(width=spec)  # E


class Overrides(Space):
    unannotated_ok = Child(width=Param())  # not a mypy error: refused at runtime (inline Param)


class Retired(Space):
    """The retired spellings of a Decision: no overload takes them."""

    entries: Boiler | HeatPump = Decision({"boiler": Boiler(kw=1)})  # E
    required: int = Decision(values=(1, 2), required=True)  # E
    named: int = Decision(values=(1, 2), name="other")  # E
    optional: Boiler | None = Decision(values=(1, 2), optional=True)  # E
